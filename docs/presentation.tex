\section{Background: Quantum Circuits, MLIR, and QIR}
%% ============================================================

\begin{frame}{What is a Quantum Circuit?}
  \begin{columns}[T]
    \column{0.52\textwidth}
      A \textbf{quantum circuit} is a program for a quantum computer, made of:
      \vspace{6pt}
      \begin{itemize}\small
        \item \hl{Qubit} --- quantum bit; can be 0, 1, or a superposition
        \item \hl{Gate} --- operation that transforms a qubit (like a logic gate)
        \item \hl{Measurement} --- reads the qubit, collapsing it to 0 or 1
      \end{itemize}
      \vspace{6pt}
      \begin{exampleblock}{Bell State --- 2 qubits in perfect correlation}
        \scriptsize
        \texttt{q0: ----H----$\bullet$----M}\\
        \texttt{\phantom{q0:} \hspace{6.5mm} |}\\
        \texttt{q1: ---------X----M}\\[2pt]
        H = Hadamard (superposition), $\bullet$--X = CNOT (entanglement)\\
        Result: always \texttt{00} or \texttt{11}, never \texttt{01} or \texttt{10}
      \end{exampleblock}
    \column{0.44\textwidth}
      \begin{center}
      \begin{tikzpicture}[
        lbl/.style={font=\scriptsize,text=white},
        gate/.style={draw=mlirblue,fill=mlirblue!30,minimum width=0.6cm,
                     minimum height=0.6cm,rounded corners=2pt,font=\scriptsize\bfseries},
        meas/.style={draw=qirgreen,fill=qirgreen!25,minimum width=0.6cm,
                     minimum height=0.6cm,rounded corners=2pt,font=\scriptsize\bfseries},
        wire/.style={thick,black},
        scale=0.90,
      ]
        % qubit wires (stop at measurement)
        \draw[wire](0.3,0)--(4.2,0);
        \draw[wire](0.3,-1.1)--(4.2,-1.1);
        % classical output (double line after measurement)
        \draw[black,double,double distance=1pt,thin](4.2,0)--(4.7,0);
        \draw[black,double,double distance=1pt,thin](4.2,-1.1)--(4.7,-1.1);
        % labels
        \node[lbl] at (0.0,0)   {q0};
        \node[lbl] at (0.0,-1.1){q1};
        % H gate on q0
        \node[gate] at (1.2,0) {H};
        % CNOT: control on q0, target on q1
        \draw[wire,mlirblue](2.4,0)--(2.4,-1.1);
        \fill[mlirblue](2.4,0) circle(3.5pt);
        \node[gate,draw=ssaorange,fill=ssaorange!35] at (2.4,-1.1){X};
        % measurements
        \node[meas] at (3.6,0)   {M};
        \node[meas] at (3.6,-1.1){M};
      \end{tikzpicture}
      \end{center}
      \vspace{6pt}
      \begin{block}{\small Classical analogy}
        \scriptsize
        \begin{tabular}{@{}ll@{}}
          Qubit     & Bit (0 or 1) \\
          Gate      & Logic gate (AND, OR\ldots) \\
          Measure   & Read a register \\
        \end{tabular}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{What is MLIR?}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{MLIR} = \textit{Multi-Level Intermediate Representation}\\[4pt]
      It is the internal language that quantum compilers use to describe
      circuits \textit{before} lowering to hardware.
      Think of it as the compiler's \textbf{source code} that humans can read.
      \vspace{6pt}
      \begin{block}{Two quantum dialects}
        \begin{itemize}\small
          \item \hl{Catalyst dialect} --- used by PennyLane\\
                operations: \texttt{quantum.custom}, \texttt{quantum.alloc}
          \item \hlq{Quake dialect} --- used by CUDA Quantum\\
                operations: \texttt{quake.h}, \texttt{quake.alloca}
        \end{itemize}
      \end{block}
      \vspace{4pt}
      Both describe the \textbf{same Bell state} circuit\\
      but with completely different syntax.
    \column{0.46\textwidth}
      \textbf{Catalyst MLIR (PennyLane):}
      \begin{lstlisting}[style=mlirstyle]
%0 = quantum.alloc( 2) : !quantum.reg
%1 = quantum.extract %0[ 0]
%out = quantum.custom
  "Hadamard"() %1 : !quantum.bit
%2 = quantum.extract %0[ 1]
%out_0:2 = quantum.custom
  "CNOT"() %out, %2
      \end{lstlisting}
      \vspace{4pt}
      \textbf{Quake MLIR (CUDA Quantum):}
      \begin{lstlisting}[style=quakestyle]
%0 = quake.alloca !quake.veq<2>
%c0 = arith.constant 0 : i64
%1 = quake.extract_ref %0[%c0]
quake.h %1 : (!quake.ref) -> ()
quake.x [%1] %3 :
  (!quake.ref, !quake.ref) -> ()
      \end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{What is QIR?}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{QIR} = \textit{Quantum Intermediate Representation}\\[4pt]
      A standard defined by the \textbf{QIR Alliance}
      (Microsoft, NVIDIA, AWS, \ldots).
      It is \textbf{LLVM IR} --- the same format used by C/C++ compilers
      --- extended with quantum operations.
      \vspace{6pt}
      \begin{exampleblock}{Why QIR matters}
        \small
        Hardware and simulators that speak QIR can run
        circuits from \textbf{any} framework ---
        PennyLane, CUDA Quantum, Qiskit, etc.
        It is the \textbf{universal adapter}.
      \end{exampleblock}
      \vspace{4pt}
      \begin{block}{QIR naming convention}
        \small
        \texttt{@\_\_quantum\_\_qis\_\_h\_\_body} = Hadamard\\
        \texttt{@\_\_quantum\_\_qis\_\_cnot\_\_body} = CNOT\\
        \texttt{@\_\_quantum\_\_qis\_\_mz\_\_body} = Measure
      \end{block}
    \column{0.46\textwidth}
      \textbf{Bell State in QIR (.ll):}
      \begin{lstlisting}[style=qirstyle]
define void @main() #0 {
entry:
  call void
    @__quantum__rt__initialize(i8* null)
  ; H on qubit 0
  call void
    @__quantum__qis__h__body(
      %Qubit* null)
  ; CNOT(0, 1)
  call void
    @__quantum__qis__cnot__body(
      %Qubit* null,
      %Qubit* inttoptr(i64 1 to %Qubit*))
  ; Measure qubit 0
  call void
    @__quantum__qis__mz__body(
      %Qubit* null, %Result* null)
  ret void
}
      \end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Step 1: Dialect Detection}
  \begin{columns}[T]
    \column{0.50\textwidth}
      The tool \textbf{automatically} recognises which MLIR format was given.
      No configuration needed.
      \vspace{6pt}
      \begin{lstlisting}[style=pystyle]
class DialectDetector:
    def detect(self, mlir_code: str):
        for dialect in self.dialects:
            if dialect.can_parse(mlir_code):
                return dialect
        raise UnsupportedDialectError()
      \end{lstlisting}
      \vspace{4pt}
      Each dialect implements \texttt{can\_parse()}:
      \begin{lstlisting}[style=pystyle]
class CatalystDialect(BaseDialect):
    def can_parse(self, code):
        return "quantum.custom" in code

class QuakeDialect(BaseDialect):
    def can_parse(self, code):
        return ("quake.alloca" in code
             or "!quake.veq" in code)
      \end{lstlisting}
      \vspace{4pt}
      \faPlus~New dialects: just subclass \texttt{BaseDialect}

\begin{frame}{Step 2: Why Parsing is Hard --- The SSA Problem}
  \begin{columns}[T]
    \column{0.50\textwidth}
      MLIR uses \textbf{SSA form} (Static Single Assignment):
      every value is assigned exactly once and gets a name
      like \texttt{\%out}, \texttt{\%1}, \texttt{\%cst}.
      \vspace{6pt}
      \textbf{Problem:} qubit indices are \textit{buried} inside
      chains of variable references.
      \vspace{6pt}
      \begin{block}{Naive approach --- fails!}
        Extract digit from variable name:\\
        \texttt{\%out\_qubits\_\textbf{1}\#0} $\Rightarrow$ qubit 1
        \vspace{2pt}\\
        \hlr{GHZ-5 result:} CNOT(0,1), CNOT(1,2), CNOT(1,3)\\
        \hlr{WRONG} --- control qubit keeps changing
      \end{block}
      \begin{exampleblock}{SSA-aware approach --- correct!}
        Trace \texttt{\%out\_qubits\_1\#0} back:\\
        CNOT\#0 $\leftarrow$ CNOT input\#0 $\leftarrow$\\
        \texttt{\%out\_qubits\_0\#0} $\leftarrow$ extract[0] =
        qubit \textbf{0}~\checkmark\\
        \hlg{Result:} CNOT(0,1), CNOT(0,2), CNOT(0,3)
      \end{exampleblock}
    \column{0.46\textwidth}
      \textbf{SSA tracking dictionaries:}
      \vspace{4pt}
      \begin{tabular}{@{}lp{3.5cm}@{}}
        \toprule
        Dictionary & Stores \\
        \midrule
        \texttt{ssa\_qubit} & \texttt{\%var} $\to$ qubit index \\
        \texttt{ssa\_constants} & \texttt{\%var} $\to$ float angle \\
        \texttt{ssa\_meas} & \texttt{\%var} $\to$ meas index \\
        \bottomrule
      \end{tabular}
      \vspace{8pt}
      \begin{center}
      \begin{tikzpicture}[
        box/.style={draw,rounded corners=2pt,minimum width=3.2cm,

      \end{center}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Step 2: Quake Dialect Parsing}
  \begin{columns}[T]
    \column{0.50\textwidth}
      The Quake dialect (CUDA Quantum) uses a different pattern:
      qubit refs are extracted via integer constants, and gates
      carry type annotations.
      \vspace{4pt}
      \begin{lstlisting}[style=quakestyle]
; Allocate 5 qubits
%0 = quake.alloca !quake.veq<5>
; Integer constant for index
%c0 = arith.constant 0 : i64
; Extract qubit-0 reference
%1 = quake.extract_ref %0[%c0]
    : (!quake.veq<5>, i64) -> !quake.ref
; Hadamard on qubit 0
quake.h %1 : (!quake.ref) -> ()
; CNOT: [ctrl] tgt notation
quake.x [%1] %3 :
    (!quake.ref, !quake.ref) -> ()
; Parametric gate
quake.rx (%cst) %1 :
    (f64, !quake.ref) -> ()
; Measurement
quake.mz %1 : (!quake.ref) -> !quake.measure
      \end{lstlisting}
    \column{0.46\textwidth}
      \textbf{Quake SSA tracking:}
      \vspace{4pt}
      \begin{tabular}{@{}lp{3.2cm}@{}}
        \toprule
        Dictionary & Stores \\
        \midrule
        \texttt{ssa\_int} & \texttt{\%c0} $\to$ 0 (int)\\
        \texttt{ssa\_float} & \texttt{\%cst} $\to$ 2.59 (float)\\

\begin{frame}[fragile]{Step 3: QIR Code Generation}
  \begin{columns}[T]
    \column{0.50\textwidth}
      The \texttt{MLIRCircuit} is converted to QIR using a template engine.
      \vspace{4pt}
      \begin{lstlisting}[style=pystyle]
def _generate_entry_function(self, circuit):
    for op_type, op_data in circuit.ordered_ops:
        if op_type == "gate":
            lines += self._gen_gate(op_data)
        elif op_type == "measurement":
            lines += self._gen_meas(op_data)
        elif op_type == "conditional":
            lines += self._gen_cond(op_data)
      \end{lstlisting}
      \vspace{4pt}
      \begin{block}{Why \texttt{ordered\_ops}?}
        \small
        Gates and measurements are \textit{interleaved} in MLIR.
        A flat ordered list is the only way to produce
        semantically correct QIR.\\
        \vspace{2pt}
        \textbf{Wrong:} all gates, then all measurements\\
        \textbf{Right:} H $\to$ CNOT $\to$ M $\to$ if-M-then-Z
      \end{block}
    \column{0.46\textwidth}
      \textbf{Gate mapping (17 gates):}
      \vspace{3pt}
      \begin{tabular}{@{}ll@{}}
        \toprule
        Source gate & QIR function \\
        \midrule
        H (Hadamard) & \texttt{h\_\_body} \\
        X, Y, Z & \texttt{x/y/z\_\_body} \\
        S, T & \texttt{s/t\_\_body} \\
        S$^\dagger$, T$^\dagger$ & \texttt{s\_adj/t\_adj\_\_body} \\
        RX, RY, RZ & \texttt{rx/ry/rz\_\_body}(angle)\\
        CNOT & \texttt{cnot\_\_body}(ctrl, tgt)\\
        CZ, SWAP & \texttt{cz/swap\_\_body} \\
        Toffoli & \texttt{ccx\_\_body}(c,c,t) \\
        \bottomrule
      \end{tabular}
      \vspace{4pt}
      \begin{exampleblock}{\small Qubit addressing in QIR}
        \scriptsize
        Qubit 0: \texttt{\%Qubit* null}\\
        Qubit 1: \texttt{\%Qubit* inttoptr(i64 1 to \%Qubit*)}\\
        Qubit n: \texttt{\%Qubit* inttoptr(i64 n to \%Qubit*)}
      \end{exampleblock}

      \textbf{Output (QIR):}
      \begin{lstlisting}[style=qirstyle]

entry:
  call void @__quantum__rt__initialize(i8* null)
  ; H on qubit 0
  call void @__quantum__qis__h__body(
    %Qubit* null)
  ; CNOT(0,1)
  call void @__quantum__qis__cnot__body(
    %Qubit* null,
    %Qubit* inttoptr (i64 1 to %Qubit*))
  ; Measure both qubits
  call void @__quantum__qis__mz__body(
    %Qubit* null, %Result* null)
  call void @__quantum__qis__mz__body(
    %Qubit* inttoptr (i64 1 to %Qubit*),
    %Result* inttoptr (i64 1 to %Result*))
  ret void
}
      \end{lstlisting}
      \vspace{2pt}
      \hlg{Gate match:} H=1, CNOT=1 $=$ H=1, CNOT=1 \checkmark\\
      \hlg{TVD:} 98.6\% $\ge$ 95\% \checkmark
  \end{columns}

\begin{frame}[fragile]{Example: Quantum Teleportation (Conditional)}
  \begin{columns}[T]
    \column{0.48\textwidth}
      \textbf{Circuit:}
      \vspace{4pt}
      \begin{lstlisting}[style=mlirstyle]
q0 --H--*-----------M(bit0)
q1 -----X---*--H---M(bit1)
q2 ---------X---[Z if bit0]
                [X if bit1]
      \end{lstlisting}
      \vspace{6pt}
      \textbf{Challenge:} the correction gates depend on
      classical measurement results.
      The parser must trace the condition through
      \textbf{5 intermediate IR operations}:
      \vspace{4pt}
      \begin{center}
      \begin{tikzpicture}[
        node/.style={font=\tiny,draw,rounded corners=2pt,
                     fill=codebg,text=white,inner sep=2pt,
                     minimum height=0.3cm},
        arr/.style={-Stealth,condred,thick},
        scale=0.88,
      ]
        \node[node](m){\texttt{quantum.measure}};
        \node[node,right=0.15cm of m](t){\texttt{tensor.from\_elements}};
        \node[node,right=0.15cm of t](c){\texttt{stablehlo.compare}};
        \node[node,below=0.2cm of c](e){\texttt{tensor.extract}};
        \node[node,left=0.15cm of e](sc){\texttt{scf.if}};
        \draw[arr](m)--(t); \draw[arr](t)--(c);
        \draw[arr](c)--(e); \draw[arr](e)--(sc);
      \end{tikzpicture}
      \end{center}
    \column{0.48\textwidth}
      \textbf{Generated conditional QIR:}
      \begin{lstlisting}[style=qirstyle]
; Read measurement 1
%c0 = call i1
  @__quantum__qis__read_result__body(
    %Result* inttoptr(i64 1 to %Result*))
br i1 %c0, label %then0, label %else0

then0:
  call void @__quantum__qis__z__body(
    %Qubit* inttoptr(i64 2 to %Qubit*))
  br label %cont0
else0:
  br label %cont0
cont0:

; Read measurement 0
%c1 = call i1
  @__quantum__qis__read_result__body(
    %Result* null)
br i1 %c1, label %then1, label %else1

then1:
  call void @__quantum__qis__x__body(
    %Qubit* inttoptr(i64 2 to %Qubit*))
  br label %cont1
      \end{lstlisting}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Getting Started}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \textbf{1. Install dependencies}
      \begin{lstlisting}[style=pystyle]
pip install -r requirements.txt
      \end{lstlisting}

      \textbf{2. Set up Ollama (for AI path)}
      \begin{lstlisting}[style=pystyle]
curl -fsSL https://ollama.com/install.sh | sh
ollama pull llama3.1:8b
      \end{lstlisting}

      \textbf{3. Initialise knowledge base (for RAG)}
      \begin{lstlisting}[style=pystyle]
python scripts/fetch_knowledge.py
python scripts/initialize_db.py
      \end{lstlisting}

      \textbf{4. Launch the web UI}
      \begin{lstlisting}[style=pystyle]
# Local only:
streamlit run src/ui/app.py

# Share on your network:
streamlit run src/ui/app.py \
  --server.address 0.0.0.0 \
  --server.port 8501
      \end{lstlisting}
    \column{0.46\textwidth}
      \textbf{5. Use the CLI (no UI needed)}
      \begin{lstlisting}[style=pystyle]
# Fast deterministic translate
python translate.py examples/mlir/bell_state.mlir

# With agentic pipeline (Ollama)
python translate.py circuit.mlir --model llama3.1-8b

# Free cloud model (no GPU)
export HF_TOKEN=hf_...
python translate.py circuit.mlir --model gpt-oss-20b

# JSON output (for scripting / CI)
python translate.py circuit.mlir --json \
  | jq '.gate_comparison.matches'
      \end{lstlisting}

      \vspace{4pt}
      \textbf{6. Generate Quake MLIR from OpenQASM3}
      \begin{lstlisting}[style=pystyle]
python example_quake/openqasm_to_quake_mlir.py
# Saves .mlir to examples/quake_mlir/
      \end{lstlisting}

      \vspace{4pt}
      \begin{block}{No Ollama? No problem}
        \small
        SSA parse + QIR generation requires \textbf{no LLM}.
        Use \texttt{gpt-oss-20b} (HF free tier) for agents
        without a local GPU.
      \end{block}
  \end{columns}
\end{frame}
